Lineage

PRD_T5_EXTERNAL_AGGNE.PURE_SNAPSHOT.PER_VEHICLEDETAIL (+PER_RISKSELECTION, PER_CLAUSEAMENDMENTMV) → PM_01 → PM_02 (SAS business rules) → PM_03 (dedupe) → PER_COM_PAYLOAD (view) → Seq 18 template (PP_ConversionConfig) → PER_COM_COVERAGES_TARGET → REQUEST_XML → DCT

Staging built by SP_POLCIY_CONVERSION_PURE_TO_WRITE_LOAD (PM_01 at line 7499, PM_02 at 8257, PM_03 at 8609). Rules verified against Carrier_PersonalProtectionBase_1_0_0_0.xml (PP package), Carrier_CoverageLibrary_Data_1_0_0_0.xml and Carrier_Package_Data_CW_1_0_0_0.xml (shared vehicle objects) — the manuscripts PL101 is generated from.

The rule that makes everything mandatory: the manuscript's RedBookDetailsRequiredCondition = VehicleInfoOverride = 1 AND IsUnderwriters. Conversion hardcodes VehicleInfoOverride = 1 for every vehicle (no RedBook keys in scope), so all RedBook detail fields — VehicleType, BodyType, EngineSize, Gross Vehicle Mass, Number of Cylinders, Drive Description, Induction Type — become required on the UW screen for every converted vehicle. Every blank or invalid value below is a potential vehicle-screen blocker.

Validation query pack (run later, read-only)

All queries are read-only against AGGNE.DEV. Run them after the ATP-10963 data rebuild (CALL AGGNE.DEV.SP_POLCIY_CONVERSION_PURE_TO_WRITE_LOAD(); + seq 17–18 re-run) so counts reflect the full vehicle population, not the pre-fix subset.

VQ-01 … VQ-14 — full pack (one script)
-- ============================================================
-- Seq 18 Motor — gap validation pack (READ-ONLY, AGGNE.DEV)
-- Run after ATP-10963 rebuild for full-population counts.
-- ============================================================

-- VQ-01 State: what the template's hardcoded NSW is hiding
SELECT STATE, COUNT(*) AS VEHICLES, COUNT(DISTINCT PUREPOLICYNUMBER) AS POLICIES
FROM AGGNE.DEV.PM_03
GROUP BY STATE ORDER BY 2 DESC;
-- Gap size = everything where STATE <> 'NSW'

-- VQ-02 YMM filter: vehicles the view still drops
SELECT COUNT(*) AS DROPPED_VEHICLES, COUNT(DISTINCT PUREPOLICYNUMBER) AS POLICIES_AFFECTED
FROM AGGNE.DEV.PM_03
WHERE NOT (YEAR<>0 AND MAKE<>'' AND MODEL<>'');

-- VQ-03 BodyType: view output vs DCT's 17 valid codes
SELECT BODYTYPE, COUNT(*) AS CNT,
       CASE WHEN BODYTYPE IN ('Bus','Convertible','Coupe','Hatchback','PrimeMover','Sedan',
                              'Utility','Van','Wagon','TowTruck','Transporter','MobilePlant',
                              'Motorcycle','Trailer','CamperVan','Caravan','Forklift')
            THEN 'VALID' WHEN COALESCE(TRIM(BODYTYPE),'')='' THEN 'BLANK (required!)'
            ELSE 'INVALID CODE' END AS DCT_CHECK
FROM AGGNE.DEV.PER_COM_PAYLOAD GROUP BY 1 ORDER BY 2 DESC;
-- companion: raw PM_03 values the view CASE will blank
SELECT BODYTYPE, COUNT(*) FROM AGGNE.DEV.PM_03
WHERE COALESCE(TRIM(BODYTYPE),'') <> '' GROUP BY 1 ORDER BY 2 DESC;

-- VQ-04 DriveDescription: captions leaking instead of codes
SELECT DRIVEDESCRIPTION, COUNT(*) AS CNT,
       CASE WHEN DRIVEDESCRIPTION IN ('4wheeldrive','Frontwheeldrive','Rearwheeldrive','N/A')
            THEN 'VALID' WHEN COALESCE(TRIM(DRIVEDESCRIPTION),'')='' THEN 'BLANK'
            ELSE 'CAPTION SENT — INVALID' END AS DCT_CHECK
FROM AGGNE.DEV.PER_COM_PAYLOAD GROUP BY 1 ORDER BY 2 DESC;

-- VQ-05 NumberOfCylinders: zeros on a 1-16 dropdown
SELECT NUMBEROFCYLINDERS, COUNT(*) AS CNT,
       CASE WHEN TRY_TO_NUMBER(NUMBEROFCYLINDERS) BETWEEN 1 AND 16 THEN 'VALID'
            ELSE 'INVALID (0/blank/other)' END AS DCT_CHECK
FROM AGGNE.DEV.PER_COM_PAYLOAD GROUP BY 1 ORDER BY 2 DESC;

-- VQ-06 InductionType: unmapped source values + blanks on a required field
SELECT T1.INDUCTIONTYPE AS PM03_VALUE, COUNT(*) AS CNT,
       CASE WHEN T1.INDUCTIONTYPE IN ('Aspirated','Supercharged','Turbocharged') THEN 'MAPS OK'
            WHEN T1.INDUCTIONTYPE = 'Turbo and supercharged' THEN 'UNMAPPED — view blanks it'
            WHEN COALESCE(TRIM(T1.INDUCTIONTYPE),'')='' THEN 'BLANK AT SOURCE'
            ELSE 'UNEXPECTED VALUE' END AS VIEW_CHECK
FROM AGGNE.DEV.PM_03 T1 GROUP BY 1 ORDER BY 2 DESC;

-- VQ-07 VoluntaryExcess: bands outside PP's option list
SELECT VOLUNTARYEXCESS, COUNT(*) AS CNT,
       CASE WHEN VOLUNTARYEXCESS IN ('0','250','500','750','1000','1250','1500')
            THEN 'VALID' ELSE 'INVALID PP OPTION' END AS DCT_CHECK
FROM AGGNE.DEV.PER_COM_PAYLOAD GROUP BY 1 ORDER BY 2 DESC;

-- VQ-08 RegistrationNumber: FormatMask.Rego violations (A-Z, 0-9 only)
SELECT COUNT(*) AS BAD_REGO,
       COUNT(DISTINCT PUREPOLICYNUMBER) AS POLICIES
FROM AGGNE.DEV.PER_COM_PAYLOAD
WHERE NOT REGEXP_LIKE(COALESCE(REGISTRATIONNUMBER,''), '^[A-Z0-9]*$');
-- sample offenders:
SELECT DISTINCT REGISTRATIONNUMBER FROM AGGNE.DEV.PER_COM_PAYLOAD
WHERE NOT REGEXP_LIKE(COALESCE(REGISTRATIONNUMBER,''), '^[A-Z0-9]*$') LIMIT 50;

-- VQ-09 ImportType: trailing-space 'N/A ' + blanks
SELECT '['||IMPORTFLAG||']' AS VALUE_BRACKETED, COUNT(*) AS CNT
FROM AGGNE.DEV.PER_COM_PAYLOAD GROUP BY 1 ORDER BY 2 DESC;

-- VQ-10 EngineSize / GVM: zero-or-blank on required fields
SELECT COUNT(*) AS TOTAL,
  SUM(CASE WHEN COALESCE(TRIM(ENGINESIZE),'')='' OR ENGINESIZE='0' THEN 1 ELSE 0 END) AS ENGINESIZE_ZERO_BLANK,
  SUM(CASE WHEN COALESCE(TRIM(GROSSVEHICLEMASS),'')='' OR GROSSVEHICLEMASS='0' THEN 1 ELSE 0 END) AS GVM_ZERO_BLANK,
  SUM(CASE WHEN VEHICLETYPE IN ('Trailer','Caravan','CamperVan','MobileEquipment','Forklift') THEN 1 ELSE 0 END) AS ENGINELESS_TYPES
FROM AGGNE.DEV.PER_COM_PAYLOAD;

-- VQ-11 AgeOfYoungestRegularDriver: blanks on a required field
SELECT COUNT(*) AS TOTAL,
  SUM(CASE WHEN COALESCE(TRIM(AGEOFYOUNGESTREGULARDRIVER),'')='' THEN 1 ELSE 0 END) AS AGE_BLANK
FROM AGGNE.DEV.PER_COM_PAYLOAD;

-- VQ-12 VehicleType: values outside DCT's 13 + defaulted blanks
SELECT VEHICLETYPE, COUNT(*) AS CNT,
       CASE WHEN VEHICLETYPE IN ('PassengerVehicle','CamperVan','Caravan','Forklift','Motorcycle',
                                 'MobileEquipment','TowTruck','Tractor','Trailer','Transporter',
                                 'HeavyVehicle','LightCommercial','HeavyCommercial')
            THEN 'VALID' ELSE 'RAW PASSTHROUGH — INVALID' END AS DCT_CHECK
FROM AGGNE.DEV.PER_COM_PAYLOAD GROUP BY 1 ORDER BY 2 DESC;
-- blanks defaulted upstream (client-confirm item):
SELECT COUNT(*) AS BLANK_DEFAULTED_TO_PASSENGER FROM AGGNE.DEV.PM_03
WHERE COALESCE(TRIM(VEHICLETYPE),'')='' AND (YEAR<>0 AND MAKE<>'' AND MODEL<>'');

-- VQ-13 RedbookCode: keys carried but never emitted
SELECT COUNT(*) AS TOTAL,
  SUM(CASE WHEN COALESCE(TRIM(MVRBVEHICLEKEY),'') <> '' THEN 1 ELSE 0 END) AS HAS_REDBOOK_KEY
FROM AGGNE.DEV.PER_COM_PAYLOAD;

-- VQ-14 Clauses: CM_*/FV_* codes flowing into PP (disabled options)
SELECT T1.DCCLAUSETYPE, COUNT(*) AS CNT,
       CASE WHEN T1.DCCLAUSETYPE LIKE 'PM%' THEN 'VALID PP CODE'
            ELSE 'DISABLED IN PP (validRef=False)' END AS DCT_CHECK
FROM AGGNE.DEV.COM_CLAUSE T1 GROUP BY 1 ORDER BY 2 DESC;
